cmake_minimum_required(VERSION 3.25)
project(hypr_float_memory VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
include(GNUInstallDirs)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
FetchContent_Declare(tomlplusplus
  URL https://github.com/marzer/tomlplusplus/archive/refs/tags/v3.4.0.tar.gz
  URL_HASH SHA256=8517f65938a4faae9ccf8ebb36631a38c1cadfb5efa85d9a72e15b9e97d25155
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(tomlplusplus)
FetchContent_Declare(readerwriterqueue
  URL https://github.com/cameron314/readerwriterqueue/archive/refs/tags/v1.0.7.tar.gz
  URL_HASH SHA256=532224ed052bcd5f4c6be0ed9bb2b8c88dfe7e26e3eb4dd9335303b059df6691
  # Populate headers without running upstream's unconditional install/CPack rules.
  SOURCE_SUBDIR _hfm_download_only
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(readerwriterqueue)
add_library(readerwriterqueue INTERFACE)
target_include_directories(readerwriterqueue INTERFACE "${readerwriterqueue_SOURCE_DIR}")
find_package(Threads REQUIRED)
add_library(memory_core STATIC src/store.cpp src/config.cpp src/writer.cpp)
target_include_directories(memory_core PUBLIC src)
target_link_libraries(memory_core PRIVATE nlohmann_json::nlohmann_json tomlplusplus::tomlplusplus readerwriterqueue PUBLIC Threads::Threads)
target_compile_options(memory_core PRIVATE -Wall -Wextra -Wpedantic)

option(BUILD_PLUGIN "Build the Hyprland plugin" ON)
if(BUILD_PLUGIN)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(HYPRLAND REQUIRED hyprland)
  add_library(hypr-float-memory MODULE src/plugin.cpp)
  set_target_properties(hypr-float-memory PROPERTIES PREFIX "")
  target_include_directories(hypr-float-memory SYSTEM PRIVATE ${HYPRLAND_INCLUDE_DIRS})
  target_compile_options(hypr-float-memory PRIVATE ${HYPRLAND_CFLAGS_OTHER} -Wall -Wextra)
  target_link_libraries(hypr-float-memory PRIVATE memory_core)
  install(TARGETS hypr-float-memory LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/hyprland)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(suite IN ITEMS store retention queue geometry)
    if(suite STREQUAL "store")
      set(test_target memory_tests)
    else()
      set(test_target ${suite}_tests)
    endif()
    add_executable(${test_target} tests/${suite}_test.cpp)
    target_link_libraries(${test_target} PRIVATE memory_core nlohmann_json::nlohmann_json)
    add_test(NAME ${test_target} COMMAND ${test_target})
    set_tests_properties(${test_target} PROPERTIES TIMEOUT 30)
  endforeach()
endif()

option(BUILD_INTEGRATION_DRIVER "Build test-only drag driver for isolated compositor" OFF)
if(BUILD_INTEGRATION_DRIVER AND BUILD_PLUGIN)
  add_library(hfm-test-driver MODULE tests/drag_driver.cpp)
  set_target_properties(hfm-test-driver PROPERTIES PREFIX "")
  target_include_directories(hfm-test-driver SYSTEM PRIVATE ${HYPRLAND_INCLUDE_DIRS})
  target_compile_options(hfm-test-driver PRIVATE ${HYPRLAND_CFLAGS_OTHER})
endif()

option(BUILD_BENCHMARKS "Build submission benchmark" OFF)
if(BUILD_BENCHMARKS)
  add_executable(memory_benchmark bench/benchmark.cpp)
  target_link_libraries(memory_benchmark PRIVATE memory_core)
endif()

# Apply host-specific optimization only to our Release targets.
option(HFM_NATIVE_ARCH "Optimize Release builds for the build machine CPU" ON)
option(HFM_ENABLE_LTO "Enable interprocedural optimization in Release builds" ON)
if(HFM_NATIVE_ARCH)
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Cross builds require HFM_NATIVE_ARCH=OFF")
  endif()
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag("-march=native" HFM_HAS_MARCH_NATIVE)
  if(NOT HFM_HAS_MARCH_NATIVE)
    message(FATAL_ERROR "Compiler does not support -march=native; use HFM_NATIVE_ARCH=OFF")
  endif()
endif()
if(HFM_ENABLE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT HFM_HAS_IPO OUTPUT HFM_IPO_ERROR LANGUAGES CXX)
  if(NOT HFM_HAS_IPO)
    message(FATAL_ERROR "IPO unavailable: ${HFM_IPO_ERROR}. Use HFM_ENABLE_LTO=OFF to disable it.")
  endif()
endif()
foreach(target IN ITEMS memory_core hypr-float-memory memory_tests retention_tests queue_tests geometry_tests memory_benchmark hfm-test-driver)
  if(TARGET ${target})
    target_compile_options(${target} PRIVATE "$<$<AND:$<CONFIG:Release>,$<CXX_COMPILER_ID:GNU,Clang,AppleClang>>:-O3>")
    if(HFM_NATIVE_ARCH)
      target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:-march=native>")
    endif()
    if(HFM_ENABLE_LTO)
      set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    endif()
  endif()
endforeach()

install(FILES LICENSE THIRD_PARTY_NOTICES.md
  DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/hypr-float-memory)
